```ts
/**
 * Today's list of repertoires (R-221): each repertoire the player trains,
 * the ones with due moves first (R-207's order), then the rest in learn
 * order (R-145), each with its due count, whether a learn session of it
 * alone would start, and its R-151 standing for the learned percentage
 * (R-177) and Mastered (R-178).
 */
import { repertoireStanding, type Standing } from './coach-progress';
import type { TrackedReview } from './move-state';
import { dueByRepertoire, learnStart, type TrainingDay } from './training-day';

export interface TodayRow {
  readonly repertoireId: string;
  /** Its due moves, as a review of it alone counts them; 0 when paused. */
  readonly due: number;
  /** A learn session of it alone would start (not paused, moves left, allowance left). */
  readonly canLearn: boolean;
  readonly standing: Standing;
}

/** The day's Reviews with how each entered training, for R-151. */
function trackedOf(
  day: Pick<TrainingDay, 'reviews'>,
  known: ReadonlySet<string>,
): Map<string, TrackedReview> {
  return new Map(
    [...day.reviews].map(([key, review]) => [
      key,
      { review, knownWithoutLearning: known.has(key) },
    ]),
  );
}

/** Would a learn session of `repertoire` alone start today? */
const canLearn = (
  day: TrainingDay,
  repertoire: TrainingDay['repertoires'][number],
): boolean =>
  repertoire.active &&
  learnStart({
    repertoires: [repertoire],
    tracked: day.tracked,
    learnedToday: day.learnedToday,
    limits: day.limits,
  }).kind === 'drill';

/**
 * `known` holds the keys whose Review entered by "I know this"
 * (the snapshot's `known`).
 */
export function todayRows(
  day: TrainingDay,
  known: ReadonlySet<string>,
  now: Date,
): TodayRow[] {
  const tracked = trackedOf(day, known);
  const due = dueByRepertoire(day, now);
  const rank = new Map(due.map((d, i) => [d.repertoireId, i]));
  const dueOf = new Map(due.map((d) => [d.repertoireId, d.due]));
  const last = due.length;
  return day.repertoires
    .map((repertoire, order) => ({ repertoire, order }))
    .sort(
      (a, b) =>
        (rank.get(a.repertoire.id) ?? last) -
          (rank.get(b.repertoire.id) ?? last) || a.order - b.order,
    )
    .map(({ repertoire }) => ({
      repertoireId: repertoire.id,
      due: dueOf.get(repertoire.id) ?? 0,
      canLearn: canLearn(day, repertoire),
      standing: repertoireStanding(repertoire.graph, tracked, now),
    }));
}
```

```ts
import type { Account } from '@mychess/domain';
import {

  type HomeForkOffer,

import { todaySummary, type TodaySummary, type TodaySummaryRow } from './training';

export interface TodayPageRow extends TodaySummaryRow {
  /** An assignment's coach, deadline and badges; null for an own repertoire. */
  readonly assignment: AssignedListItem | null;
}

export interface TodayPage {
  readonly displayName: string;
  readonly summary: TodaySummary;
  readonly rows: readonly TodayPageRow[];
  readonly offers: readonly HomeForkOffer[];
}

/** Each row with its assignment's facts, by repertoire id. */
export function joinFacts(
  rows: readonly TodaySummaryRow[],
  assigned: readonly AssignedListItem[],
): TodayPageRow[] {
  const byId = new Map(assigned.map((a) => [a.repertoireId, a]));
  return rows.map((row) => ({
    ...row,
    assignment: row.kind === 'assigned' ? (byId.get(row.repertoireId ?? row.id) ?? null) : null,
  }));
}

export async function todayPage(account: Account): Promise<TodayPage> {
  const [summary, { fromCoach, offers }] = await Promise.all([
    todaySummary(account),
    homeAssignments(),
  ]);
  return {
    displayName: account.displayName,
    summary,
    rows: joinFacts(summary.rows, fromCoach),
    offers,
  };
}
```

```ts
// Which of the hero card's four states Today shows (spec, "The hero card").
export type HeroState = 'due' | 'learn' | 'done' | 'empty';

export function heroState({
  due,
  toLearn,
  rows,
}: {
  readonly due: number;
  readonly toLearn: number;
  /** How many repertoires the player trains. */
  readonly rows: number;
}): HeroState {
  if (rows === 0) return 'empty';
  if (due > 0) return 'due';
  return toLearn > 0 ? 'learn' : 'done';
}
```

```tsx
// Buttons on the accent card: the main one inverted (surface on accent),
// the second outlined in the card's text colour. 44px targets.
const HERO_BUTTON =
  'inline-flex min-h-touch items-center justify-center rounded-control px-4 py-2 text-base font-semibold';
const HERO_PRIMARY = `${HERO_BUTTON} bg-surface text-accent`;
const HERO_SECONDARY = `${HERO_BUTTON} border border-accent-fg text-accent-fg`;

function NextReview({ next }: { readonly next: Date | null }): JSX.Element | null {
  const t = useTranslations('training.today');
  const reviewTime = useReviewTime();
  return next === null ? null : <p>{t('next', { when: reviewTime(next) })}</p>;
}

function Lead({ state, summary }: { readonly state: HeroState; readonly summary: TodaySummary }): JSX.Element {
  const t = useTranslations();
  if (state === 'due')
    return (
      <>
        <p className="text-3xl font-bold">{t('today.dueCount', { count: summary.due })}</p>
        <p>{t('today.minutes', { minutes: summary.minutes })}</p>
      </>
    );
  if (state === 'empty') return <p className="text-lg">{t('today.welcome')}</p>;
  return (
    <>
      <p className="text-xl font-semibold">
        {state === 'learn' ? t('training.today.nothingDue') : t('today.allDone')}
      </p>
      <NextReview next={summary.nextDue} />
    </>
  );
}

function Actions({ state, summary }: { readonly state: HeroState; readonly summary: TodaySummary }): JSX.Element | null {
  const t = useTranslations();
  const learn = (
    <Link href={trainingPath('learn')} className={state === 'due' ? HERO_SECONDARY : HERO_PRIMARY}>
      {t('training.today.learn', { count: summary.toLearn })}
    </Link>
  );
  if (state === 'due')
    return (
      <div className="flex flex-wrap gap-3">
        <Link href={trainingPath('review')} className={HERO_PRIMARY}>{t('training.today.start')}</Link>
        {summary.toLearn > 0 ? learn : null}
      </div>
    );
  if (state === 'learn') return <div className="flex flex-wrap gap-3">{learn}</div>;
  if (state === 'empty')
    return (
      <div className="flex flex-wrap gap-3">
        <Link href={ROUTES.repertoires} className={HERO_PRIMARY}>{t('today.create')}</Link>
        <Link href={ROUTES.repertoireImport} className={HERO_SECONDARY}>{t('today.import')}</Link>
      </div>
    );
  return null;
}

/** Today's hero card (spec, "The hero card"; R-142, R-148). */
export function TodayHero({ state, summary }: { readonly state: HeroState; readonly summary: TodaySummary }): JSX.Element {
  const t = useTranslations('training.today');
  const titleId = useId();
  return (
    <section aria-labelledby={titleId} className="flex flex-col gap-3 rounded-card bg-accent p-5 text-accent-fg shadow-card">
      <h2 id={titleId} className="text-sm font-semibold uppercase tracking-wide">{t('title')}</h2>
      <Lead state={state} summary={summary} />
      <Actions state={state} summary={summary} />
    </section>
  );
}
```

```tsx
'use client';

import { percentLearned } from '@mychess/chess';
import { useTranslations } from 'next-intl';
import Link from 'next/link';
import { useId, type JSX } from 'react';
import { assignedRepertoirePath, repertoirePath, trainingPath } from '../../lib/routes';
import type { TodayPageRow } from '../../server/today-page';
import { AssignmentFacts } from '../assignment-facts';
import { MasteredBadge } from '../coach/state-bar';
import { CARD } from '../surfaces';
import { LearnedBar } from './learned-bar';

const ROW_BUTTON =
  'inline-flex min-h-touch shrink-0 items-center justify-center rounded-control bg-accent-subtle px-4 py-2 font-semibold text-accent';

function RowAction({ row, describedBy }: { readonly row: TodayPageRow; readonly describedBy: string }): JSX.Element | null {
  const t = useTranslations('today');
  if (row.due > 0)
    return (
      <Link href={trainingPath('review', row.id)} aria-describedby={describedBy} className={ROW_BUTTON}>
        {t('review', { count: row.due })}
      </Link>
    );
  if (!row.canLearn) return null;
  return (
    <Link href={trainingPath('learn', row.id)} aria-describedby={describedBy} className={ROW_BUTTON}>
      {t('learn')}
    </Link>
  );
}

function Progress({ row }: { readonly row: TodayPageRow }): JSX.Element | null {
  const t = useTranslations('coach.progress');
  const percent = percentLearned(row.counts);
  if (percent === null) return null;
  return (
    <span className="flex flex-col gap-1">
      <LearnedBar percent={percent} />
      <span className="flex flex-wrap items-center gap-2 text-sm text-fg-muted">
        {t('percent', { percent })}
        {row.mastered ? <MasteredBadge /> : null}
      </span>
    </span>
  );
}

function RepertoireRow({ row }: { readonly row: TodayPageRow }): JSX.Element {
  const t = useTranslations('repertoires.create');
  const nameId = useId();
  const href = row.kind === 'assigned' ? assignedRepertoirePath(row.id) : repertoirePath(row.id);
  return (
    <li className="flex items-start gap-3 border-b border-border py-3 last:border-b-0">
      <span className="flex min-w-0 flex-1 flex-col gap-1">
        <span className="flex flex-wrap items-baseline gap-x-2">
          <Link id={nameId} href={href} className="break-words font-semibold text-fg underline-offset-2 hover:underline">
            {row.name}
          </Link>
          <span className="text-sm text-fg-muted">{t(row.side)}</span>
        </span>
        {row.assignment === null ? null : (
          <AssignmentFacts assignment={row.assignment} className="gap-1 text-sm" />
        )}
        <Progress row={row} />
      </span>
      <RowAction row={row} describedBy={nameId} />
    </li>
  );
}

/** Today's list of every repertoire the player trains (R-221, R-222). */
export function RepertoireRows({ rows }: { readonly rows: readonly TodayPageRow[] }): JSX.Element {
  const t = useTranslations('today');
  const titleId = useId();
  return (
    <section aria-labelledby={titleId} className={`${CARD} flex flex-col p-5`}>
      <h2 id={titleId} className="text-xl font-bold">{t('repertoires')}</h2>
      <ul>
        {rows.map((row) => <RepertoireRow key={row.id} row={row} />)}
      </ul>
    </section>
  );
}
```
